CMAKE_MINIMUM_REQUIRED(VERSION 3.9)
CMAKE_POLICY(SET CMP0079 NEW)

PROJECT(B3)

SET(CMAKE_EXPORT_COMPILE_COMMANDS ON)

SET(CMAKE_CXX_STANDARD 20)
SET(CMAKE_CXX_STANDARD_REQUIRED ON)

SET(CMAKE_C_STANDARD 17)
SET(CMAKE_C_STANDARD_REQUIRED ON)

IF(NOT CMAKE_BUILD_TYPE)

    OPTION(B33_RUN_TESTS "Run tests before starting the main loop" ON)
    SET(CMAKE_BUILD_TYPE "Debug")

ENDIF()

IF(CMAKE_BUILD_TYPE STREQUAL "Debug")

    OPTION(B33_DEBUG_BUILD "Enables debug build" ON)

ENDIF()

SET(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/Bin/${CMAKE_SYSTEM_PROCESSOR}/${CMAKE_BUILD_TYPE})
SET(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/Lib/${CMAKE_SYSTEM_PROCESSOR}/${CMAKE_BUILD_TYPE})
SET(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/Lib/${CMAKE_SYSTEM_PROCESSOR}/${CMAKE_BUILD_TYPE})

ADD_SUBDIRECTORY(${CMAKE_CURRENT_SOURCE_DIR}/Engine)
ADD_SUBDIRECTORY(${CMAKE_CURRENT_SOURCE_DIR}/Projects)

FOREACH( B33_TARGET ${B33_TARGETS} )

    IF (MSVC)
    
        SET_TARGET_PROPERTIES(${B33_TARGET} PROPERTIES
            WIN32_EXECUTABLE TRUE
        )
    
        TARGET_COMPILE_OPTIONS(${B33_TARGET} PUBLIC "/permissive-")
        TARGET_COMPILE_OPTIONS(${B33_TARGET} PUBLIC "/GR")
    
        IF (CMAKE_BUILD_TYPE!="Debug")
    
            TARGET_COMPILE_OPTIONS(${B33_TARGET} PUBLIC "/O2")
    
        ENDIF()
    
    ENDIF()
    
    IF (UNIX AND NOT APPLE)
    
        IF (CMAKE_BUILD_TYPE STREQUAL "Debug")
    
            TARGET_COMPILE_OPTIONS(${B33_TARGET} PRIVATE "-g")
    
        ELSE()
    
            TARGET_COMPILE_OPTIONS(${B33_TARGET} PRIVATE "-O3")
    
        ENDIF()
    
    ENDIF()
    
    IF (APPLE)
    
        IF (CMAKE_BUILD_TYPE STREQUAL "Debug")
    
            TARGET_COMPILE_OPTIONS(${B33_TARGET} PRIVATE "-g")
    
        ELSE()
    
            TARGET_COMPILE_OPTIONS(${B33_TARGET} PUBLIC "-O3")
    
        ENDIF()
    
    ENDIF()
    
    TARGET_COMPILE_DEFINITIONS(${B33_TARGET} PUBLIC ${B33_COMPILE_DEFINITIONS})
    
    TARGET_LINK_LIBRARIES(${B33_TARGET} PUBLIC
        B33Core
        B33App
        B33Rendering
        B33System
        Jolt
    )
    
    MESSAGE("Built project ${B33_TARGET}.")
    MESSAGE("Build type ${CMAKE_BUILD_TYPE}.")
    MESSAGE("Compile definitions ${B33_COMPILE_DEFINITIONS}.")

ENDFOREACH()
