CMAKE_MINIMUM_REQUIRED(VERSION 3.15)

PROJECT(B33)

SET(CMAKE_EXPORT_COMPILE_COMMANDS ON)

SET(CMAKE_CXX_STANDARD 20)
SET(CMAKE_CXX_STANDARD_REQUIRED ON)

SET(CMAKE_C_STANDARD 17)
SET(CMAKE_C_STANDARD_REQUIRED ON)

SET(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/Bin/${CMAKE_SYSTEM_PROCESSOR}/${CMAKE_BUILD_TYPE})
SET(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/Lib/${CMAKE_SYSTEM_PROCESSOR}/${CMAKE_BUILD_TYPE})
SET(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/Lib/${CMAKE_SYSTEM_PROCESSOR}/${CMAKE_BUILD_TYPE})

IF(NOT TARGET_ARCH OR TARGET_ARCH STREQUAL "")

    SET(TARGET_ARCH "unknown")

ENDIF()

IF(NOT CMAKE_BUILD_TYPE OR (CMAKE_BUILD_TYPE STREQUAL "Debug"))

    ADD_COMPILE_DEFINITIONS(_DEBUG)

ENDIF()

IF(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")

    ADD_COMPILE_DEFINITIONS(_MSCV)

ENDIF()

IF(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")

    ADD_COMPILE_DEFINITIONS(_CLANG)

ENDIF()

IF(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")

    ADD_COMPILE_DEFINITIONS(_GNU)

ENDIF()

ADD_COMPILE_DEFINITIONS(UNICODE)
ADD_COMPILE_DEFINITIONS(_UNICODE)
ADD_COMPILE_DEFINITIONS(_CRT_SECURE_NO_WARNINGS)

ADD_SUBDIRECTORY(${CMAKE_CURRENT_SOURCE_DIR}/Engine)
ADD_SUBDIRECTORY(${CMAKE_CURRENT_SOURCE_DIR}/Projects)

IF (MSVC)

    SET_TARGET_PROPERTIES(${B33TARGET} PROPERTIES
        WIN32_EXECUTABLE TRUE
    )

ENDIF()

IF (MSVC)

    TARGET_COMPILE_OPTIONS(${B33TARGET} PUBLIC "/permissive-")
    TARGET_COMPILE_OPTIONS(${B33TARGET} PUBLIC "/GR")

    IF (CMAKE_BUILD_TYPE!="Debug")

        TARGET_COMPILE_OPTIONS(${B33TARGET} PUBLIC "/O2")

    ENDIF()

ENDIF()

IF (UNIX AND NOT APPLE)

    IF (CMAKE_BUILD_TYPE STREQUAL "Debug")

        TARGET_COMPILE_OPTIONS(${B33TARGET} PRIVATE "-g")

    ELSE()

        TARGET_COMPILE_OPTIONS(${B33TARGET} PRIVATE "-O3")

    ENDIF()

ENDIF()

IF (APPLE)

    IF (CMAKE_BUILD_TYPE STREQUAL "Debug")

        TARGET_COMPILE_OPTIONS(${B33TARGET} PRIVATE "-g")

    ELSE()

        TARGET_COMPILE_OPTIONS(${B33TARGET} PUBLIC "-O3")

    ENDIF()

ENDIF()

TARGET_LINK_LIBRARIES(${B33TARGET} PRIVATE
    B33Core
    B33App
    B33Rendering
    B33System
    Jolt
)
